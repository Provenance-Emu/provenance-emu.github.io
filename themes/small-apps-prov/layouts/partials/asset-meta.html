{{- /* Classifies a release asset (from data/builds.json) by its filename for the /install/ downloads list. Returns: platform key, platformLabel, platformIcon, arch label, archTitle, and a friendly kind label (falls back to the filename). Order matters: tvOS before iOS, x86_64 before x86. */ -}} {{- $n := lower .name -}} {{- $platform := "other" -}} {{- if strings.Contains $n "tvos" -}}{{- $platform = "tvos" -}} {{- else if or (strings.HasSuffix $n ".ipa") (strings.Contains $n "ios") -}}{{- $platform = "ios" -}} {{- else if or (strings.Contains $n "macos") (strings.Contains $n "darwin") (strings.HasSuffix $n ".dmg") -}}{{- $platform = "macos" -}} {{- else if or (strings.Contains $n "windows") (strings.Contains $n "win64") (strings.Contains $n "win32") (strings.HasSuffix $n ".exe") (strings.HasSuffix $n ".msi") -}}{{- $platform = "windows" -}} {{- else if or (strings.Contains $n "linux") (strings.HasSuffix $n ".appimage") (strings.HasSuffix $n ".deb") (strings.HasSuffix $n ".rpm") -}}{{- $platform = "linux" -}} {{- else if or (strings.Contains $n "android") (strings.HasSuffix $n ".apk") -}}{{- $platform = "android" -}} {{- end -}} {{- $platforms := dict "ios" (dict "label" "iOS / iPadOS" "icon" "fas fa-mobile-alt") "tvos" (dict "label" "tvOS" "icon" "fas fa-tv") "macos" (dict "label" "macOS" "icon" "fab fa-apple") "windows" (dict "label" "Windows" "icon" "fab fa-windows") "linux" (dict "label" "Linux" "icon" "fab fa-linux") "android" (dict "label" "Android" "icon" "fab fa-android") "other" (dict "label" "Other" "icon" "fas fa-box") -}} {{- $p := index $platforms $platform -}} {{- $arch := "" -}}{{- $archTitle := "" -}} {{- if strings.Contains $n "universal" -}}{{- $arch = "Universal" -}}{{- $archTitle = "Apple silicon and Intel" -}} {{- else if or (strings.Contains $n "aarch64") (strings.Contains $n "arm64") -}}{{- $arch = "ARM64" -}}{{- $archTitle = "64-bit ARM (Apple silicon, Snapdragon, Raspberry Pi 4/5…)" -}} {{- else if or (strings.Contains $n "x86_64") (strings.Contains $n "amd64") (strings.Contains $n "x64") -}}{{- $arch = "x64" -}}{{- $archTitle = "64-bit Intel / AMD" -}} {{- else if or (strings.Contains $n "armv7") (strings.Contains $n "armhf") -}}{{- $arch = "ARM · 32-bit" -}}{{- $archTitle = "32-bit ARM (ARMv7)" -}} {{- else if or (strings.Contains $n "i386") (strings.Contains $n "i686") (strings.Contains $n "x86") (strings.Contains $n "win32") -}}{{- $arch = "x86 · 32-bit" -}}{{- $archTitle = "32-bit Intel / AMD" -}} {{- end -}} {{- $kind := .name -}} {{- if and (strings.HasSuffix $n ".ipa") (strings.Contains $n "trollstore") -}}{{- $kind = "IPA · TrollStore" -}} {{- else if strings.HasSuffix $n ".ipa" -}}{{- $kind = cond (strings.Contains $n "noextensions") "IPA · no extensions" "IPA" -}} {{- else if and (strings.Contains $n "libretro") (or (strings.HasSuffix $n ".dylib") (strings.HasSuffix $n ".so") (strings.HasSuffix $n ".dll")) -}}{{- $kind = "RetroArch core" -}} {{- else if strings.HasSuffix $n ".appimage" -}}{{- $kind = "AppImage" -}} {{- else if strings.HasSuffix $n ".deb" -}}{{- $kind = ".deb package" -}} {{- else if strings.HasSuffix $n ".rpm" -}}{{- $kind = ".rpm package" -}} {{- else if strings.HasSuffix $n ".msi" -}}{{- $kind = "Installer (.msi)" -}} {{- else if and (strings.HasSuffix $n ".exe") (strings.Contains $n "setup") -}}{{- $kind = "Installer" -}} {{- else if strings.HasSuffix $n ".exe" -}}{{- $kind = ".exe" -}} {{- else if strings.HasSuffix $n ".dmg" -}}{{- $kind = "Disk image (.dmg)" -}} {{- else if strings.HasSuffix $n ".zip" -}}{{- $kind = cond (eq $platform "windows") "Portable .zip" (cond (eq $platform "macos") "App (.zip)" ".zip") -}} {{- end -}} {{- return dict "platform" $platform "platformLabel" $p.label "platformIcon" $p.icon "arch" $arch "archTitle" $archTitle "kind" $kind -}}
